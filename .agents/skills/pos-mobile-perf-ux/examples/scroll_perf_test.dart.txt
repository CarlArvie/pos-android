// Needs dev_dependencies: integration_test (sdk: flutter), flutter_driver (sdk: flutter). Ask before adding.
// Give the product grid widget `key: const Key('product_grid')` and seed data via --dart-define=SEED_ROWS.
import 'package:flutter/material.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:integration_test/integration_test.dart';
import 'package:pos/main.dart' as app; // package name comes from pubspec `name: pos`

void main() {
  final binding = IntegrationTestWidgetsFlutterBinding.ensureInitialized();

  testWidgets('product grid scroll performance', (tester) async {
    app.main();
    await tester.pumpAndSettle();

    final grid = find.byKey(const Key('product_grid'));
    expect(grid, findsOneWidget);

    await binding.traceAction(() async {
      for (var i = 0; i < 5; i++) {
        await tester.fling(grid, const Offset(0, -600), 4000);
        await tester.pumpAndSettle();
        await tester.fling(grid, const Offset(0, 600), 4000);
        await tester.pumpAndSettle();
      }
    }, reportKey: 'scroll_summary');
  });
}
